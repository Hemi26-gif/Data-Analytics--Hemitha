-- ============================================================
-- 01. OPERATOR NAME
-- Create this as a CALCULATED COLUMN in dim_operators
-- Modeling / Table tools -> New column
-- ============================================================

Operator Name =
TRIM(
    dim_operators[first_name]
        & " "
        & dim_operators[last_name]
)


-- ============================================================
-- 02. PRODUCTION PERFORMANCE
-- Use these measures for Operator-wise Production and
-- Shift-wise Production & Defects visuals.
-- Modeling -> New measure
-- ============================================================

Total Production =
SUM(
    fact_production_runs[quantity_produced]
)


Total Defects =
SUM(
    fact_production_runs[quantity_defective]
)


Production Runs =
COUNTROWS(
    fact_production_runs
)


Average Output per Run =
DIVIDE(
    [Total Production],
    [Production Runs],
    0
)


-- ============================================================
-- 03. OPERATOR & SHIFT ANALYSIS
-- Use Operator Name or shift from dim_operators as the axis.
-- ============================================================

Active Operators =
DISTINCTCOUNT(
    fact_production_runs[operator_id]
)


Average Operator Experience =
AVERAGE(
    dim_operators[experience_years]
)


Defect Rate =
DIVIDE(
    [Total Defects],
    [Total Production],
    0
)


-- ============================================================
-- 04. DOWNTIME ANALYSIS
-- Use for the Operator Downtime visual.
-- ============================================================

Total Downtime =
SUM(
    fact_production_runs[downtime_minutes]
)


Average Downtime per Run =
DIVIDE(
    [Total Downtime],
    [Production Runs],
    0
)


-- ============================================================
-- 05. OPTIONAL PAGE 7 KPI MEASURES
-- These are useful for cards or tooltips.
-- ============================================================

Average Defects per Run =
DIVIDE(
    [Total Defects],
    [Production Runs],
    0
)


Production per Active Operator =
DIVIDE(
    [Total Production],
    [Active Operators],
    0
)
